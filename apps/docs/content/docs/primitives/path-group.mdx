## Usage

<CodeVariants>
<Variant framework="react">

```tsx
<Remold.PathGroup
  viewBox="0 0 200 200"
  paths={[
    { id: "face", d: face, className: "face" },
    { id: "extra", d: mood.extra, className: "extra" }, // a tear, or blushing cheeks
    { id: "brows", d: mood.brows, fill: "none", stroke: "currentColor", strokeWidth: 7 },
    { id: "eyes", d: mood.eyes, className: "eyes" },
    { id: "mouth", d: mood.mouth, className: "mouth" },
  ]}
/>
```

</Variant>
<Variant framework="js">

```ts
import { createMorph, value } from "@remold-ui/core";
import { frames } from "@remold-ui/dom";

// One path element per part, each morphing on its own spring.
const parts = {
  brows: document.querySelector("#brows"),
  eyes: document.querySelector("#eyes"),
  mouth: document.querySelector("#mouth"),
};

function show(mood: Record<keyof typeof parts, string>) {
  for (const [id, path] of Object.entries(parts)) {
    const morph = createMorph(path.getAttribute("d")!, mood[id]);
    const t = value(0, { spring: { duration: 0.6, bounce: 0.2 } });
    t.to(1);
    frames((dt) => {
      t.step(dt);
      path.setAttribute("d", morph.d(t.current));
      return !t.settled;
    });
  }
}
```

</Variant>
</CodeVariants>

Each part is matched by its `id` and morphs from wherever it is when its path changes, even into a shape with a different number of pieces: angry brows lift into arches, a frown turns into a smile and opens into a laugh, and two round eyes become two hearts. Parts are drawn in the order you list them, so the eyes and mouth stay in front of the face.

Style the parts with `className` or SVG attributes. Colours aren't morphed by Remold; a CSS `transition` on `fill` and `stroke` keeps them in step, as in the example.
